    elements = {"nsamp":  {"schema": U32},

                "tx_id":  {"schema": U16},

                "cfg_id": {"schema": U16}}

 

class MmFir(FreeRunMod):          # free-running kernel

    mm_views = (                  # what a host sees

        RegBank("regs", cfg_port="s_cfg", ...),

        QueueIn("qin", port="s_in", depth=64),

        QueueOut("qout", port="m_out", depth=64))

    proc_latency: int = 9         # measured at RTL

 

    def run_iter(self):           # behavior + timing

        hdr = yield from self.s_in.get_schema(FirCmdHdr)

        x, t0 = yield from self.s_in.get_pipelined(

                S16, hdr.nsamp)

        y = self._filter(x.val)   # numpy golden

        yield from self.m_out.write_pipelined(

                array(S64, y), t_out)
